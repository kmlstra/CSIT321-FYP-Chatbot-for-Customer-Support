# 🤖 RASA TRAINING & REPORTING COMMANDS
# Navigate to backend directory before running these commands

# ===== TRAINING COMMANDS =====

# 1. Basic Training
python -m rasa train

# 2. Force Retrain
python -m rasa train --force

# 3. Train with Config
python -m rasa train --config config.yml --domain domain.yml --data data

# ===== TESTING & ACCURACY COMMANDS =====

# 4. Quick Accuracy Test (no retraining)
python -m rasa test nlu --nlu data/nlu --out results/quick_accuracy

# 5. Reliable Accuracy Test (trains 5 times for better accuracy)
python -m rasa test nlu --nlu data/nlu --cross-validation --folds 5 --out results/reliable_accuracy

# 5b. Faster Reliable Test (trains 3 times - good for FYP)
python -m rasa test nlu --nlu data/nlu --cross-validation --folds 3 --out results/fyp_accuracy

# 6. Story/Dialogue Test
python -m rasa test --stories data/stories

# 7. Comprehensive Test
python -m rasa test --out results/comprehensive_test

# ===== DETAILED REPORTING COMMANDS =====

# 8. Generate Confusion Matrix
python -m rasa test nlu --nlu data/nlu --out results/confusion_matrix

# 9. NLU Evaluation Report
python -m rasa test nlu --nlu data/nlu --cross-validation --out results/nlu_evaluation

# 10. Story Evaluation Report
python -m rasa test --stories data/stories --out results/story_evaluation --e2e

# 11. Complete Evaluation
python -m rasa test --out results/complete_evaluation

# ===== INTERACTIVE TESTING =====

# 12. Interactive Shell
python -m rasa shell

# 13. Interactive Shell (Debug)
python -m rasa shell --debug

# ===== SERVER COMMANDS =====

# 14. Start RASA Server
python -m rasa run --enable-api --cors "*" --port 5005

# 15. Start Actions Server
python -m rasa run actions --actions api.actions.rasa_actions --port 5055

# ===== QUICK WORKFLOWS =====

# 16. Train + Test Workflow
python -m rasa train && python -m rasa test nlu --cross-validation

# 17. Complete Workflow
python -m rasa train && python -m rasa test --out results/final_report

# ===== MOST IMPORTANT FOR FYP =====

# Train and get accuracy (recommended workflow)
python -m rasa train && python -m rasa test nlu --cross-validation --folds 5 --out results/accuracy_report

# Quick accuracy check after training
python -m rasa test nlu --cross-validation --out results/quick_accuracy
